    if name_value != "C1_marker":
        return
    var row: Dictionary = cases[-1]
    var probe := SpatialProbe.new()
    root.add_child(probe)
    var result: Dictionary = await probe.run_probe(self,row["snapshot"],row["board"],"lucky-glyph",out_dir,"C1_marker")

    result["eligibility_source"] = "Unchanged original _candidate(marker,0): pre-generation canonical Crew trust, production environment_test_result and publication. No placement, label or membership injection after generation."

    file.store_string(JSON.stringify(result,"  "))
    file.close()
    probe.free()
